cmake_minimum_required(VERSION 3.22)
project(IgniteMusicalCore LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(staged "${CMAKE_CURRENT_BINARY_DIR}/original-model")
file(GLOB original_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/upstream/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/stage.py" "${CMAKE_CURRENT_SOURCE_DIR}/provenance.json" ${original_sources})
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/stage.py" "${staged}" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
    message(FATAL_ERROR "Original model staging failed")
endif()
add_library(ignite_original_patterns "${staged}/Pattern.cpp" "${staged}/Bar.cpp" "${staged}/Step.cpp" "${staged}/SequenceManager.cpp")
target_compile_features(ignite_original_patterns PUBLIC cxx_std_17)
target_include_directories(ignite_original_patterns PUBLIC "${staged}" "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(original_patterns_test tests/patterns_test.cpp)
target_link_libraries(original_patterns_test PRIVATE ignite_original_patterns)
enable_testing()
add_test(NAME original_patterns COMMAND original_patterns_test)
add_executable(original_sequence_test tests/sequence_test.cpp)
target_link_libraries(original_sequence_test PRIVATE ignite_original_patterns)
add_test(NAME original_sequence COMMAND original_sequence_test)
add_library(ignite_original_playback src/PatternPlayback.cpp)
target_link_libraries(ignite_original_playback PUBLIC ignite_original_patterns)
add_executable(original_playback_test tests/playback_test.cpp)
target_link_libraries(original_playback_test PRIVATE ignite_original_playback)
add_test(NAME original_playback COMMAND original_playback_test "${CMAKE_CURRENT_SOURCE_DIR}/../Standalone/resources/Ignite.sf2" "${CMAKE_CURRENT_BINARY_DIR}/original-core-audio.wav")
