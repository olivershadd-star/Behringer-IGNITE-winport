cmake_minimum_required(VERSION 3.22)
project(IgniteCoreDesktop VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory(../DesktopCore original-core)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/stage_effects.py")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/stage_effects.py" "${CMAKE_CURRENT_BINARY_DIR}/effects" RESULT_VARIABLE result)
if(NOT result EQUAL 0)
    message(FATAL_ERROR "Original effects staging failed")
endif()
juce_add_gui_app(IgniteCoreDesktop PRODUCT_NAME "IgniteCoreDesktop" COMPANY_NAME "Ignite Community")
add_library(ignite_original_effects "${CMAKE_CURRENT_BINARY_DIR}/effects/TrackEffects.cpp")
target_include_directories(ignite_original_effects PUBLIC src PRIVATE ../Standalone/vendor/ignite)
target_link_libraries(ignite_original_effects PUBLIC ignite_original_playback juce::juce_dsp)
add_executable(original_effects_test tests/effects_test.cpp)
target_link_libraries(original_effects_test PRIVATE ignite_original_effects)
add_test(NAME original_effects COMMAND original_effects_test)
target_sources(IgniteCoreDesktop PRIVATE src/Main.cpp ../Standalone/src/app/EmbeddedBank.cpp ../Standalone/src/app/WindowsAudioOutput.cpp ../Standalone/src/app/WindowsInput.cpp)
target_include_directories(IgniteCoreDesktop PRIVATE src ../Standalone/src/app ../Standalone/vendor/ignite)
target_link_libraries(IgniteCoreDesktop PRIVATE ignite_original_effects juce::juce_gui_extra juce::juce_recommended_config_flags)
target_compile_definitions(IgniteCoreDesktop PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_FLAC=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
if(MSVC)
    target_compile_options(IgniteCoreDesktop PRIVATE /utf-8)
    target_compile_options(ignite_original_effects PRIVATE /utf-8)
endif()
if(WIN32)
    enable_language(RC)
    file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../Standalone/resources/Ignite.sf2" IGNITE_BANK)
    configure_file(src/Bank.rc.in "${CMAKE_CURRENT_BINARY_DIR}/Bank.rc" @ONLY)
    target_sources(IgniteCoreDesktop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/Bank.rc")
    target_link_libraries(IgniteCoreDesktop PRIVATE user32 winmm avrt)
endif()
