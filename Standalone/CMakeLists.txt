cmake_minimum_required(VERSION 3.22)
project(IgniteStandalone VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(ignite_audio_core src/core/AudioCore.cpp)
target_include_directories(ignite_audio_core PUBLIC src/core)
add_executable(ignite_core_tests tests/core_tests.cpp)
target_link_libraries(ignite_core_tests PRIVATE ignite_audio_core)
enable_testing()
add_test(NAME audio_core COMMAND ignite_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/resources/Ignite.sf2" "${CMAKE_CURRENT_BINARY_DIR}/ignite-demo.wav")

option(IGNITE_BUILD_APP "Build the JUCE Windows application" ON)
if(IGNITE_BUILD_APP)
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 7.0.12 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
    juce_add_gui_app(IgniteStandalone PRODUCT_NAME "IgniteStandalone" COMPANY_NAME "Ignite Community")
    target_sources(IgniteStandalone PRIVATE src/app/Main.cpp src/app/MainComponent.cpp src/app/PatternFile.cpp src/app/WindowsInput.cpp)
    target_link_libraries(IgniteStandalone PRIVATE ignite_audio_core juce::juce_audio_utils juce::juce_gui_extra juce::juce_recommended_config_flags)
    target_compile_definitions(IgniteStandalone PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_FLAC=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
    if(WIN32)
        target_link_libraries(IgniteStandalone PRIVATE user32)
    endif()
    add_custom_command(TARGET IgniteStandalone POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/resources/Ignite.sf2" "$<TARGET_FILE_DIR:IgniteStandalone>/Ignite.sf2"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/resources/GeneralUser-GS-LICENSE.txt" "$<TARGET_FILE_DIR:IgniteStandalone>/GeneralUser-GS-LICENSE.txt")
endif()
